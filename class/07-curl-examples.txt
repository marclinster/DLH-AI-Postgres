-- send a request to Mistral's embedding API
-- use the model codestral-embed and ask for a 6-dimensional embedding

curl -X POST "https://api.mistral.ai/v1/embeddings" \
     -H "Content-Type: application/json" \
     -H "Authorization: Bearer ${MISTRAL_API_KEY}" \
     -d '{"model": "codestral-embed", "input": ["Embed this sentence."], "output_dimension":6}'


-- store the mebedding in the file embedding6.json

curl -X POST "https://api.mistral.ai/v1/embeddings" \
     -H "Content-Type: application/json" \
     -H "Authorization: Bearer ${MISTRAL_API_KEY}" \
     -d '{"model": 
          "codestral-embed", 
          "input": ["Embed this sentence."], 
          "output_dimension":6}' \
     -o embedding6.json   

-- use jq to inspect that file

jq '.' embedding6.json
jq '.id' embedding6.json
jq '.data[0]' embedding6.json
jq '.data[0].embedding' embedding6.json
jq '.data[0].embedding[1]' embedding6.json
